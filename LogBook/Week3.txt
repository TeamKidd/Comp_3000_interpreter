---squares.lox
var number_squares = 100;

print "clearscreen";

fun draw_line(size){
    for(var j = 0; j < size; j = j + 1){
        print "fd 1";
    }
}


fun create_square(size){
    draw_line(size);
    print "rt 90";
    draw_line(size);
    print "rt 90";
    draw_line(size);
    print "rt 90";
    draw_line(size);
    print "rt 90";
}

for (var i = 0; i < number_squares; i = i + 1){
    create_square(i);
    draw_line(i);
    print "rt 15";
}
